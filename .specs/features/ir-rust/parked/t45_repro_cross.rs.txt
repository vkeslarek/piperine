//! Repro (not a committed test): a `CROSS` watch fires at the first accepted point after the
//! sign change, not at the located crossing (docs/spec part_v §17.3 rules 1 and 3; abi.md §4.3
//! "located in the step"). 1 V 1 kHz sine vs a 0.5 V reference: the true upward crossing is at
//! asin(0.5) / (2 pi 1 kHz) = 83.333 us; the expression at the firing must be `0 < e < etol`
//! (etol = 1e-6 V, the voltage nature abstol) or the firing within ttol of the true time (ttol is a session option with no stated default; 1 ns is far above the 1p the spec examples use).
#![cfg(unix)]
#[allow(dead_code)]
#[path = "../../piperine-solver/tests/helpers/abi_devices.rs"]
mod abi_devices;
#[allow(dead_code)]
#[path = "../../piperine-solver/tests/helpers/abi_mixed.rs"]
mod abi_mixed;

use std::cell::RefCell;
use std::rc::Rc;

use abi_devices::{add_capacitor, add_resistor, add_vsrc, node_id, vsine};
use abi_mixed::comparator;
use piperine_solver::abi::{ParamValue, Port};
use piperine_solver::prelude::{CircuitBuilder, Solver, TransientAnalysisOptions};
use piperine_solver::protocol::{Observation, ObservationKind, Sink};

struct Collect(Rc<RefCell<Vec<Observation>>>);
impl Sink for Collect {
    fn observe(&mut self, o: &Observation) { self.0.borrow_mut().push(o.clone()); }
}

#[test]
fn repro() {
    let mut b = CircuitBuilder::new("crossing");
    let gnd = b.ground();
    let top = b.node("top");
    let out = b.node("out");
    b.device(vsine(), "v1", &[("amp", ParamValue::Real(1.0)), ("freq", ParamValue::Real(1.0e3))], &[Port::Node(top.clone()), Port::Node(gnd.clone())]);
    add_resistor(&mut b, "r1", 1.0e3, &top, &out);
    add_capacitor(&mut b, "c1", 1.0e-6, &out, &gnd);
    let r = b.node("r");
    add_vsrc(&mut b, "vr", 0.5, &r, &gnd);
    b.device(comparator(), "cmp", &[], &[Port::Node(top.clone()), Port::Node(r), Port::Net("out".into())]);
    let seen = Rc::new(RefCell::new(Vec::new()));
    // a step cap of 1 us: the grid the crossing could be located on
    let mut options = TransientAnalysisOptions::new(2.0e-4, 1.0e-6);
    options.dt_max = 1.0e-6;
    let mut solver = Solver::new(b.build().expect("build")).with_tran_opts(options).with_sink(Box::new(Collect(Rc::clone(&seen)))).build();
    let result = solver.tran().expect("tran").solve().expect("solve");
    let seen = seen.borrow();
    let fired = seen.iter().find(|o| o.kind == ObservationKind::Crossing && o.path == "cmp.up").expect("the crossing fired");
    let v = result.iter().find(|s| (s.time() - fired.time).abs() < 1e-15).and_then(|s| s.get_node(&node_id(&top))).expect("an accepted point");
    let truth = (0.5f64).asin() / (2.0 * std::f64::consts::PI * 1.0e3);
    println!("fired at {:e} s, true crossing {:e} s, v(top) at the firing {:e} V (e = {:e} V)", fired.time, truth, v, v - 0.5);
    assert!(v - 0.5 < 1.0e-6 || (fired.time - truth).abs() < 1.0e-9 + 1.0e-15, "late by {:e} s with e = {:e} V > etol", fired.time - truth, v - 0.5);
}
