//! Repro (not a committed test): two commits to one net in one microstep; the settled value is
//! the one before, so `changed()` must not fire (docs/spec part_iii §8.2.1, part_iv §12.1.2).
#![cfg(unix)]
#[allow(dead_code)]
#[path = "../../piperine-solver/tests/helpers/abi_mixed.rs"]
mod abi_mixed;
#[allow(dead_code)]
#[path = "../../piperine-solver/tests/helpers/partials.rs"]
mod partials;

use std::cell::Cell as StdCell;
use std::result::Result;
use std::sync::{Arc, OnceLock};

use abi_mixed::delay;
use piperine_abi::device::{Device, Eval, Fault, Flow, React, Values};
use piperine_abi::layout::{BIT, HIGH, LOW};
use piperine_abi::raw::{Cell, Module, Shape, Statement, cell, part, pin_flag, role};
use piperine_abi::shape::{FunctionSpec, PinSpec, ReactionSpec, ShapeBuf, ShapeRef};
use piperine_abi::thunk::module;
use piperine_solver::abi::{ParamValue, Port};
use piperine_solver::prelude::{CircuitBuilder, Solver, TransientAnalysisOptions};

thread_local! { static FIRED: StdCell<u32> = const { StdCell::new(0) }; }

fn net(name: &str) -> Port { Port::Net(name.to_string()) }

/// pins: x=0 (PORT BIT, init LOW); go=1 (PORT event)
/// reaction 0: on [go] writes [S0 {x}, S1 {x}]; commits HIGH then LOW, no delay
struct TwoWrites;
static TW_SHAPE: OnceLock<ShapeRef> = OnceLock::new();
static TW: OnceLock<Module> = OnceLock::new();
fn tw_shape() -> ShapeRef {
    let mut s = ShapeBuf::new();
    let x = s.pin(PinSpec { flags: pin_flag::HAS_INIT, init: &LOW, ..PinSpec::discrete("x", role::PORT, &BIT) });
    let go = s.pin(PinSpec::event("go", role::PORT));
    let st = Statement { pin: x, offset: 0, len: 2, flags: 0, min_after: 0.0 };
    s.reaction(ReactionSpec { name: "two", origin: "repro", on: vec![go], writes: vec![st, st], ..ReactionSpec::default() });
    s.freeze()
}
impl Device for TwoWrites {
    fn prototype() -> &'static Shape<'static> { TW_SHAPE.get_or_init(tw_shape).shape() }
    fn create(_: &Values<'_>) -> Result<(Self, ShapeRef), Fault> { Ok((TwoWrites, Arc::clone(TW_SHAPE.get_or_init(tw_shape)))) }
    fn eval(&self, _k: u32, _cx: &mut Eval) -> Result<(), Fault> { Ok(()) }
    fn react(&self, _k: u32, cx: &mut React) -> Result<Flow, Fault> {
        cx.commit(0, &HIGH, 0.0)?;
        cx.commit(1, &LOW, 0.0)?;
        Ok(Flow::Continue)
    }
}

/// pins: x=0 (PORT BIT); flip=1 (PRIVATE event); fn 0 reads [x.CHANGED] writes [flip.CHANGED];
/// reaction 0: on [flip] counts
struct Watch;
static W_SHAPE: OnceLock<ShapeRef> = OnceLock::new();
static W: OnceLock<Module> = OnceLock::new();
fn w_shape() -> ShapeRef {
    let mut s = ShapeBuf::new();
    let x = s.pin(PinSpec::discrete("x", role::PORT, &BIT));
    let flip = s.pin(PinSpec::event("flip", role::PRIVATE));
    s.function(FunctionSpec {
        name: "edge",
        origin: "repro",
        reads: vec![Cell { kind: cell::PIN, index: x, part: part::CHANGED }],
        writes: vec![Cell { kind: cell::PIN, index: flip, part: part::CHANGED }],
        ..FunctionSpec::default()
    });
    s.reaction(ReactionSpec { name: "count", origin: "repro", on: vec![flip], ..ReactionSpec::default() });
    s.freeze()
}
impl Device for Watch {
    fn prototype() -> &'static Shape<'static> { W_SHAPE.get_or_init(w_shape).shape() }
    fn create(_: &Values<'_>) -> Result<(Self, ShapeRef), Fault> { Ok((Watch, Arc::clone(W_SHAPE.get_or_init(w_shape)))) }
    fn eval(&self, _k: u32, cx: &mut Eval) -> Result<(), Fault> {
        let changed = cx.bytes(0) == [1];
        cx.set(0, changed as u8);
        Ok(())
    }
    fn react(&self, _k: u32, _cx: &mut React) -> Result<Flow, Fault> {
        FIRED.with(|f| f.set(f.get() + 1));
        Ok(Flow::Continue)
    }
}

#[test]
fn repro() {
    let mut b = CircuitBuilder::new("repro");
    b.device(delay(), "kick", &[("at", ParamValue::Real(1.0e-7))], &[net("go")]);
    b.device(TW.get_or_init(|| module::<TwoWrites>("TwoWrites")), "tw", &[], &[net("x"), net("go")]);
    b.device(W.get_or_init(|| module::<Watch>("Watch")), "w", &[], &[net("x")]);
    let mut solver = Solver::new(b.build().expect("build")).with_tran_opts(TransientAnalysisOptions::new(2.0e-7, 1.0e-8)).build();
    solver.tran().expect("tran").solve().expect("solve");
    // x is LOW before, LOW after (the last commit wins): it never changed
    let fired = FIRED.with(|f| f.get());
    println!("x.changed() fired {fired} time(s): once for the initial value at (0, 0) (part_iv 617), and the expected total is 1");
    assert_eq!(fired, 1, "x.changed() fired although the settled value is the one it had before the microstep");
}
