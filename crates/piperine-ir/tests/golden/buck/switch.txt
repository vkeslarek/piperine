nature Electrical
  potential Voltage v "V" abstol 1e-6
  flow Current i "A" abstol 1e-12
body Switch
  param ron: f64 = 0.02 where (ron > 0.0)
  param roff: f64 = 1000000.0 where (roff > ron)
  node p: Electrical port
  node n: Electrical port
  carrier g: bool in
  branch b0 (p, n) flow
  analog
    b0.flow +> direct (probe(p, n) * if g { (1.0 / ron) } else { (1.0 / roff) })
body Top
instance Top: Top  nodes ()  params ()  top
